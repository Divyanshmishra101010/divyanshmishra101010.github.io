from langchain.chains import RetrievalQAWithSourcesChain
from langchain.chat_models import ChatOpenAI
from langchain.vectorstores import Chroma
from langchain.embeddings import OpenAIEmbeddings
from langchain.tools import Tool

# Initialize embedding model and vector store (ChromaDB)
embeddings = OpenAIEmbeddings(model="text-embedding-3-large")
vectordb = Chroma(persist_directory="./chroma_db", embedding_function=embeddings)

# MMR retriever for diverse, high-quality context chunks
retriever = vectordb.as_retriever(
    search_type="mmr",
    search_kwargs={"k": 6, "fetch_k": 20}
)

# Custom tool: Yahoo Finance + FMP real-time market data
def fetch_market_data(ticker: str) -> dict:
    """Fetches real-time price, sector, valuation from yfinance."""
    import yfinance as yf
    return yf.Ticker(ticker).info

market_tool = Tool(
    name="MarketDataFetcher",
    func=fetch_market_data,
    description="Get real-time financial data for a stock ticker."
)

llm = ChatOpenAI(model="gpt-4o", temperature=0.1, streaming=True)
rag_chain = RetrievalQAWithSourcesChain.from_chain_type(
    llm=llm, retriever=retriever, return_source_documents=True
)
from fastapi import FastAPI, BackgroundTasks, Depends
from fastapi.responses import StreamingResponse
from redis.asyncio import Redis
import hashlib, json

app = FastAPI(title="Financial AI Backend", version="2.0.0")

@app.post("/api/v1/portfolio/chat")
async def portfolio_chat(req: ChatRequest, redis: Redis = Depends(get_redis)):
    """Agentic RAG with Redis semantic cache and SSE streaming."""
    cache_key = hashlib.md5(req.query.encode()).hexdigest()
    cached = await redis.get(cache_key)
    if cached:
        return json.loads(cached)

    async def stream_tokens():
        async for chunk in rag_chain.astream({"question": req.query}):
            yield f"data: {chunk}\n\n"

    return StreamingResponse(stream_tokens(), media_type="text/event-stream")

@app.post("/api/v1/portfolio/rebalance")
async def trigger_rebalance(req: RebalanceRequest, bg: BackgroundTasks):
    """Async rebalancing trigger. 2+ hrs reduced to 2 seconds."""
    bg.add_task(run_rebalancing_engine, req.tenant_id, req.model_id)
    return {"status": "queued", "eta_seconds": 2}
from pydantic import BaseModel, Field
from typing import Literal
from uuid import UUID

class ChatRequest(BaseModel):
    query: str = Field(..., min_length=3)
    session_id: UUID
    tenant_id: str
    temperature: float = Field(default=0.1, ge=0.0, le=1.0)

class RebalanceRequest(BaseModel):
    tenant_id: str
    model_id: str
    account_ids: list[str]
    strategy: Literal["aggressive", "moderate", "conservative"] = "moderate"

class TradeProposal(BaseModel):
    ticker: str
    action: Literal["BUY", "SELL"]
    quantity: float
    estimated_value: float
    rebalance_delta_pct: float = Field(ge=-100.0, le=100.0)
