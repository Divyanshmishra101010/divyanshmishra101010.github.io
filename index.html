from langchain.chains import RetrievalQAWithSourcesChain
from langchain_openai import ChatOpenAI, OpenAIEmbeddings
from langchain_community.vectorstores import Chroma
from langchain.tools import Tool

# 1. High-dimensional embedding engine and ChromaDB vector store
embeddings = OpenAIEmbeddings(model="text-embedding-3-large")
vectordb = Chroma(persist_directory="./chroma_db", embedding_function=embeddings)

# 2. Maximal Marginal Relevance (MMR) retriever for high context diversity
retriever = vectordb.as_retriever(
    search_type="mmr",
    search_kwargs={"k": 6, "fetch_k": 20}
)

# 3. Custom Tool Execution: Live Yahoo Finance & FMP market intelligence
def fetch_security_metrics(ticker: str) -> dict:
    """Fetches real-time price, sector, and valuation from financial APIs."""
    import yfinance as yf
    stock = yf.Ticker(ticker)
    return {
        "price": stock.info.get("currentPrice"),
        "sector": stock.info.get("sector"),
        "pe_ratio": stock.info.get("trailingPE")
    }

market_tool = Tool(
    name="MarketDataTool",
    func=fetch_security_metrics,
    description="Retrieve real-time financial security metrics for a given ticker."
)

llm = ChatOpenAI(model="gpt-4o", temperature=0.1, streaming=True)
rag_chain = RetrievalQAWithSourcesChain.from_chain_type(
    llm=llm, retriever=retriever, return_source_documents=True
)
from fastapi import FastAPI, BackgroundTasks, Depends, HTTPException
from fastapi.responses import StreamingResponse
from redis.asyncio import Redis
import hashlib, json

app = FastAPI(title="Financial AI & Portfolio Engine", version="2.0.0")

@app.post("/api/v1/portfolio/chat/stream")
async def stream_portfolio_analysis(req: ChatRequest, redis: Redis = Depends(get_redis)):
    """Agentic RAG endpoint with Redis semantic cache & SSE token streaming."""
    cache_key = f"rag:{req.tenant_id}:{hashlib.md5(req.query.encode()).hexdigest()}"
    cached = await redis.get(cache_key)
    if cached:
        return json.loads(cached)

    async def token_generator():
        async for chunk in rag_chain.astream({"question": req.query}):
            yield f"data: {chunk}\n\n"

    return StreamingResponse(token_generator(), media_type="text/event-stream")

from pydantic import BaseModel, Field
from typing import Literal
from uuid import UUID

class ChatRequest(BaseModel):
    tenant_id: str
    session_id: UUID
    query: str = Field(..., min_length=3, description="User prompt or question")
    temperature: float = Field(default=0.1, ge=0.0, le=1.0)

class RebalanceRequest(BaseModel):
    tenant_id: str
    model_id: str
    account_ids: list[str] = Field(..., min_items=1)
    strategy: Literal["aggressive", "moderate", "conservative"] = "moderate"

class TradeProposal(BaseModel):
    ticker: str
    action: Literal["BUY", "SELL"]
    quantity: float = Field(..., gt=0)
    estimated_value: float
    rebalance_delta_pct: float = Field(ge=-100.0, le=100.0)
